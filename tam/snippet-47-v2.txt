<script>
(function(){
  var MA=["menu-item-11551","menu-item-11552"];
  function bung(){
    var m=document.getElementById("dov-spmega");
    if(!m || m.querySelector(".dov-huongdan")) return;
    var hang=document.createElement("div");
    hang.className="dov-huongdan";
    MA.forEach(function(ma){
      var a=document.querySelector("#menu-item-221 ."+ma+" > a");
      if(!a) return;
      var b=document.createElement("a");
      b.href=a.getAttribute("href");
      b.textContent=a.textContent.trim();
      hang.appendChild(b);
    });
    if(hang.children.length) m.insertBefore(hang, m.firstChild);
  }
  if(document.readyState!=="loading") bung();
  document.addEventListener("DOMContentLoaded", bung);
  setTimeout(bung,600); setTimeout(bung,2000);
})();
</script>
